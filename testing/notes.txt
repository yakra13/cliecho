def get_internal_args(user_yaml, schema):
    internal_data = {}
    
    # Loop through the fields defined in the tool's schema
    for field_name, metadata in schema['properties'].items():
        if field_name in user_yaml:
            # Map the user's value to the internal standard name
            internal_key = metadata.get('x-internal-type', field_name)
            internal_data[internal_key] = user_yaml[field_name]
            
    return internal_data
